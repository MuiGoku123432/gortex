## CI structural checker

Both task verifies extract this block from this file into a temporary script and run it with the workflow paths as arguments (it needs `yq` v4 and `python3`, both present on the developer machine).

```python ci-check
import json, os, re, subprocess, sys

STEP = "Configure read access to the private COBOL parser module"
STOCK = "Stock COBOL grammar must fail closed"
FORK = "github.com/MuiGoku123432/tree-sitter-cobol-upgrade"
GO = re.compile(r"\bgo (build|test|mod|install|run|vet|list|generate)\b|goreleaser-cross|make claude-plugin")
bad = []


def uses(step):
    return str(step.get("uses", ""))


for path in sys.argv[1:]:
    wf = json.loads(subprocess.check_output(["yq", "-o=json", path]))
    if (wf.get("env") or {}).get("GOPRIVATE") != FORK:
        bad.append(f"{path}: workflow-level env GOPRIVATE is not {FORK}")
    if "pull_request_target" in json.dumps(wf.get("on")):
        bad.append(f"{path}: pull_request_target trigger")
    for name, job in (wf.get("jobs") or {}).items():
        steps = job.get("steps") or []
        setup = [i for i, s in enumerate(steps) if uses(s).startswith("actions/setup-go@")]
        builds = any(
            GO.search(str(s.get("run", "")))
            or uses(s).startswith(("golangci/golangci-lint-action@", "golang/govulncheck-action@"))
            for s in steps
        )
        if builds and not setup:
            bad.append(f"{path}:{name}: builds Go without an actions/setup-go step")
        for i in setup:
            if (steps[i].get("with") or {}).get("cache") is not False:
                bad.append(f"{path}:{name}: setup-go cache is not false")
        for s in steps:
            if uses(s).startswith("golang/govulncheck-action@") and (s.get("with") or {}).get("cache") is not False:
                bad.append(f"{path}:{name}: govulncheck-action cache is not false")
        if not setup:
            continue
        nxt = setup[0] + 1
        cred = steps[nxt] if nxt < len(steps) else {}
        run = str(cred.get("run", ""))
        if cred.get("name") != STEP:
            bad.append(f"{path}:{name}: '{STEP}' does not immediately follow setup-go")
            continue
        if cred.get("shell") != "bash":
            bad.append(f"{path}:{name}: credential step is not shell: bash")
        if "secrets.COBOL_PARSER_READ_PAT" not in json.dumps(cred.get("env")):
            bad.append(f"{path}:{name}: credential step does not take the secret from step env")
        if "::error::" not in run or "exit 1" not in run or "insteadOf" not in run:
            bad.append(f"{path}:{name}: credential step must fail with ::error:: when unset and write an insteadOf rewrite")
        if re.search(r"echo[^\n]*\$\{?COBOL_PARSER_READ_PAT", run):
            bad.append(f"{path}:{name}: credential step echoes the secret")
    base = os.path.basename(path)
    if base == "ci.yml":
        stock = [s for s in (wf["jobs"]["test"].get("steps") or []) if s.get("name") == STOCK]
        if not stock:
            bad.append(f"{path}:test: missing '{STOCK}' step")
        else:
            s = stock[0]
            if s.get("if") != "matrix.os == 'ubuntu-latest'":
                bad.append(f"{path}:test: stock step is not ubuntu-only")
            if "-dropreplace" not in s.get("run", "") or "compiled forest language identity mismatch" not in s.get("run", ""):
                bad.append(f"{path}:test: stock step must drop the forest replace and require the attestation mismatch text")
    if base == "release.yml":
        steps = wf["jobs"]["release"].get("steps") or []
        dock = [i for i, s in enumerate(steps) if "goreleaser-cross" in str(s.get("run", ""))]
        dl = [i for i, s in enumerate(steps) if re.search(r"\bgo mod download\b", str(s.get("run", "")))]
        if not dock or not dl or dl[0] > dock[0]:
            bad.append(f"{path}:release: host 'go mod download' must run before the goreleaser-cross container")
        if dock:
            s = steps[dock[0]]
            for tok in ("GOPROXY=off", "/go/pkg/mod", "GOMODCACHE", "-e GOPRIVATE", "-e GITHUB_TOKEN"):
                if tok not in s.get("run", ""):
                    bad.append(f"{path}:release: container step lacks {tok}")
            if "COBOL_PARSER_READ_PAT" in json.dumps(s):
                bad.append(f"{path}:release: fork credential reaches the container step")
            if set((s.get("env") or {}).keys()) != {"GITHUB_TOKEN"}:
                bad.append(f"{path}:release: container step env must be GITHUB_TOKEN only")

print("\n".join(bad) if bad else "OK")
sys.exit(1 if bad else 0)
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Repository secrets to job steps | `COBOL_PARSER_READ_PAT` is exposed to specific steps. |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-02-20 | Information Disclosure | `COBOL_PARSER_READ_PAT` | high | mitigate | Fine-grained single-repository Contents read-only token (provisioned in 02-06); passed only through step `env`; never echoed (the checker rejects echoing it; GitHub masks secrets); no `pull_request_target`; fork PRs do not receive secrets. |
| T-02-21 | Information Disclosure | Actions cache | high | mitigate | `cache: false` on every setup-go step and on govulncheck-action, enforced by the checker, so private module source is never saved to a restorable cache. |
| T-02-22 | Information Disclosure | goreleaser-cross container | high | mitigate | Modules are downloaded on the host; the container gets a read-from-cache module path, `GOPROXY=off`, `GOPRIVATE`, and only `GITHUB_TOKEN`; the checker rejects the fork secret anywhere in that step. |
| T-02-23 | Tampering | release module set | medium | mitigate | Default read-only module mode plus `GOPROXY=off` means the release builds exactly the go.sum-pinned module set, or fails loudly. |
| T-02-24 | Information Disclosure | runner `~/.gitconfig` | low | accept | The insteadOf rewrite holding the token lives only for the job on an ephemeral GitHub-hosted runner. |
| T-02-25 | Information Disclosure | golangci-lint results cache | low | accept | golangci-lint analyzes only this module's packages; dependency source is not linted or cached by it. |
| T-02-SC | Tampering | third-party actions | medium | mitigate | The new setup-go step reuses the repository's existing commit-pinned `actions/setup-go@b7ad1dad...`; no new action is introduced. |
